{{- with $logo }}{{ end -}} {{ $org | safeHTML }}{{ with $role }} · {{ . | safeHTML }}{{ end }} {{- with $dates }}{{ . | safeHTML }}{{ end -}}

{{- else -}} {{ .Text | safeHTML }} {{- end -}}
